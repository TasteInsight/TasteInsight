# 数据库开发规范文档

# 一、目标

统一数据库设计和操作规范，提高数据库性能和可维护性，确保数据安全和一致性。


# 二、数据库规约

## 2.1 命名规约

1. 【强制】Prisma 模型命名使用 PascalCase 单数形式，数据库表名使用复数形式

```prisma
// 正确示例 - TasteInsight 项目 schema.prisma

// ==================== 用户相关模型 ====================

model User {
  id        String   @id @default(cuid())
  openId    String   @unique
  nickname  String
  avatar    String?
  allergens String[] @default([]) // 过敏原列表

  // 关联关系
  preferences    UserPreference?
  settings       UserSetting?
  reviews        Review[]
  comments       Comment[]
  mealPlans      MealPlan[]
  uploads        DishUpload[]
  reports        Report[]
  favoriteDishes FavoriteDish[]
  browseHistory  BrowseHistory[]
  aiSessions     AISession[]

  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  @@map("users")
}

model UserPreference {
  id     String @id @default(cuid())
  userId String @unique
  user   User   @relation(fields: [userId], references: [id], onDelete: Cascade)

  // 基础偏好
  tagPreferences String[] @default([]) // 菜系偏好
  priceMin       Float    @default(0)
  priceMax       Float    @default(50)

  // 口味偏好
  spicyLevel Int @default(0) // 0-5 辣度等级
  sweetness  Int @default(0) // 0-5
  saltiness  Int @default(0) // 0-5
  oiliness   Int @default(0) // 0-5

  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  @@map("user_preferences")
}

// ==================== 菜品相关模型 ====================

model Dish {
  id          String   @id @default(cuid())
  name        String
  tags        String[]
  price       Float
  priceUnit   String?
  description String?
  images      String[]

  // 口味信息
  spicyLevel Int @default(0) // 0-5 辣度等级
  sweetness  Int @default(0) // 0-5 甜度
  saltiness  Int @default(0) // 0-5 咸度
  oiliness   Int @default(0) // 0-5 油度

  // 位置信息
  canteenId    String
  canteen      Canteen @relation(fields: [canteenId], references: [id], onDelete: Cascade)
  canteenName  String
  windowId     String?
  window       Window? @relation(fields: [windowId], references: [id])
  windowName   String

  // 供应时间
  availableMealTime String[] // breakfast, lunch, dinner, nightsnack

  // 状态
  status String @default("online") // online, offline

  // 统计信息
  averageRating Float @default(0)
  reviewCount   Int   @default(0)

  // 关联关系
  reviews       Review[]
  favoritedBy   FavoriteDish[]
  mealPlans     MealPlanDish[]
  browseHistory BrowseHistory[]

  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  @@index([canteenId])
  @@index([windowId])
  @@index([status])
  @@index([averageRating])
  @@map("dishes")
}

// 错误示例
model users {  // 错误：模型名应使用单数PascalCase
  id String @id
}

model Dish {
  id String @id
  @@map("dish")  // 错误：表名应使用复数
}
```

1. 【强制】字段命名使用 camelCase，数据库列名使用 snake_case

```
// 正确示例
model Order {
  id            String   @id @default(uuid())
  orderNumber   String   @unique @map("order_number")
  totalAmount   Decimal  @map("total_amount") @db.Decimal(10, 2)
  shippingFee   Decimal  @default(0) @map("shipping_fee") @db.Decimal(10, 2)
  orderStatus   String   @default("pending") @map("order_status")
  createdAt     DateTime @default(now()) @map("created_at")
  updatedAt     DateTime @updatedAt @map("updated_at")
  
  @@map("orders")
  @@index([orderNumber])
  @@index([orderStatus])
}

// 错误示例
model Order {
  id String @id
  order_number String  // 错误：Prisma字段应使用camelCase
  OrderStatus String   // 错误：应使用camelCase
}
```

1. 【推荐】关联字段使用有意义的名称

```
// 正确示例
```

## 2.2 字段设计

1. 【强制】主键使用 cuid() 或 uuid()，避免使用自增 ID

```prisma
// 正确示例 - TasteInsight 项目使用 cuid()
model User {
  id String @id @default(cuid())  // 使用cuid
  
  @@map("users")
}

model Dish {
  id String @id @default(cuid())
  
  @@map("dishes")
}

// 不推荐（仅特殊场景使用）
model User {
  id Int @id @default(autoincrement())  // 自增ID可能暴露数据量
  
  @@map("users")
}
```

1. 【强制】所有表必须包含 createdAt 和 updatedAt 字段


1. 【推荐】使用分页查询处理大量数据

```typescript
// TasteInsight 菜品列表分页实现
// backend/src/dishes/dishes.service.ts
async getDishes(
  getDishesDto: GetDishesDto,
  userId: string,
): Promise<DishListResponseDto> {
  const { filter, search, sort, pagination } = getDishesDto;

  // 构建 where 条件
  const andConditions: Prisma.DishWhereInput[] = [];
  
  // 状态筛选：默认只显示 online 的菜品
  if (!filter.includeOffline) {
    andConditions.push({ status: 'online' });
  }
  
  // 评分、价格、标签等筛选条件...
  
  const where: Prisma.DishWhereInput =
    andConditions.length > 0 ? { AND: andConditions } : {};

  // 排序配置
  const orderBy: Prisma.DishOrderByWithRelationInput = {};
  if (sort.field) {
    orderBy[sort.field] = sort.order || SortOrder.ASC;
  } else {
    orderBy.averageRating = SortOrder.DESC;
  }

  // 分页计算
  const skip = (pagination.page - 1) * pagination.pageSize;
  const take = pagination.pageSize;

  // 并行查询数据和总数
  const [items, total] = await Promise.all([
    this.prisma.dish.findMany({
      where,
      orderBy,
      skip,
      take,
      include: {
        canteen: true,
        window: true,
      },
    }),
    this.prisma.dish.count({ where }),
  ]);

  const totalPages = Math.ceil(total / pagination.pageSize);

  return {
    code: 200,
    message: 'success',
    data: {
      items: items.map((dish) => DishDto.fromEntity(dish)),
      meta: {
        page: pagination.page,
        pageSize: pagination.pageSize,
        total,
        totalPages,
      },
    },
  };
}

// 更优 - 基于游标的分页（适合大数据集）
async function getCursorPaginatedDishes(cursor?: string, take: number = 20) {
  const dishes = await prisma.dish.findMany({
    take: take + 1,
    ...(cursor && {
      cursor: { id: cursor },
      skip: 1
    }),
    orderBy: {
      averageRating: 'desc'
    },
    include: {
      canteen: true,
      window: true,
    }
  })
  
  const hasMore = dishes.length > take
  const data = hasMore ? dishes.slice(0, -1) : dishes
  const nextCursor = hasMore ? data[data.length - 1].id : undefined
  
  return {
    data,
    nextCursor,
    hasMore
  }
}
```

```typescript
// TasteInsight 评论创建事务示例
// backend/src/comments/comments.service.ts
async createComment(
  userId: string,
  dto: CreateCommentDto,
): Promise<CommentResponseDto> {
  return this.prisma.$transaction(async (tx) => {
    // 使用悲观锁锁定 Review 记录，防止并发导致楼层号重复
    const lockedReviews =
      await tx.$queryRaw`SELECT id FROM reviews WHERE id = ${dto.reviewId} FOR UPDATE`;

    if (!Array.isArray(lockedReviews) || lockedReviews.length === 0) {
      throw new NotFoundException('评价不存在或未通过审核');
    }

    const review = await tx.review.findUnique({
      where: { id: dto.reviewId, status: 'approved', deletedAt: null },
    });
    if (!review) {
      throw new NotFoundException('评价不存在或未通过审核');
    }

    // 检查父评论
    if (dto.parentCommentId) {
      const parent = await tx.comment.findUnique({
        where: { id: dto.parentCommentId },
      });
      if (!parent || parent.reviewId !== dto.reviewId) {
        throw new NotFoundException('父评论不存在');
      }
    }

    // 计算楼层号
    const count = await tx.comment.count({
      where: { reviewId: dto.reviewId },
    });
    const floor = count + 1;

    // 创建评论
    const comment = await tx.comment.create({
      data: {
        reviewId: dto.reviewId,
        userId,
        content: dto.content,
        parentCommentId: dto.parentCommentId,
        status: autoApprove ? 'approved' : 'pending',
        floor,
      },
    });

    return comment;
  });
}
```

```typescript
// TasteInsight 菜品缓存示例
// backend/src/dishes/dishes.service.ts
@Injectable()
export class DishesService {
  constructor(
    private prisma: PrismaService,
    private redis: RedisService
  ) {}
  
  async findById(id: string): Promise<Dish | null> {
    const cacheKey = `dish:${id}`
    
    // 先从缓存获取
    const cached = await this.redis.get(cacheKey)
    if (cached) {
      return JSON.parse(cached)
    }
    
    // 从数据库查询
    const dish = await this.prisma.dish.findUnique({
      where: { id },
      include: {
        canteen: true,
        window: true,
      }
    })
    
    // 缓存结果
    if (dish) {
      await this.redis.set(
        cacheKey,
        JSON.stringify(dish),
        'EX',
        3600  // 1小时过期
      )
    }
    
    return dish
  }
  
  async update(id: string, data: UpdateDishDto): Promise<Dish> {
    const dish = await this.prisma.dish.update({
      where: { id },
      data
    })
    
    // 更新后清除缓存
    await this.redis.del(`dish:${id}`)
    
    return dish
  }
}
```

```typescript
@Injectable()
export class CleanupService {
  private readonly logger = new Logger(CleanupService.name)
  
  constructor(private prisma: PrismaService) {}
  
  @Cron('0 2 * * *')  // 每天凌晨2点执行
  async cleanupOldData() {
    this.logger.log('开始清理过期数据')
    
    const thirtyDaysAgo = new Date()
    thirtyDaysAgo.setDate(thirtyDaysAgo.getDate() - 30)
    
    try {
      // 清理过期的浏览历史
      const deletedHistory = await this.prisma.browseHistory.deleteMany({
        where: {
          viewedAt: {
            lt: thirtyDaysAgo
          }
        }
      })
      
      this.logger.log(`清理了 ${deletedHistory.count} 条过期浏览记录`)
      
      // 清理已软删除超过30天的评价
      const deletedReviews = await this.prisma.review.deleteMany({
        where: {
          deletedAt: {
            lt: thirtyDaysAgo
          }
        }
      })
      
      this.logger.log(`永久删除了 ${deletedReviews.count} 条评价`)
      
      // 清理过期的管理员会话
      const deletedSessions = await this.prisma.adminSession.deleteMany({
        where: {
          expiresAt: {
            lt: new Date()
          }
        }
      })
      
      this.logger.log(`清理了 ${deletedSessions.count} 个过期管理员会话`)
    } catch (error) {
      this.logger.error('清理数据失败', error)
    }
  }
```
